    62→function initials(name: string): string {
    63→  const parts = name.trim().split(/\s+/).filter(Boolean);
    64→  if (parts.length === 0) return "?";
    65→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
    66→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
    67→}

   519→export function UserMenuButton({ onOpenManage, onOpenManageRoles }: MenuProps) {
   520→  const qc = useQueryClient();
   521→  const [loginTarget, setLoginTarget] = useState<PublicUser | null>(null);
   522→  const [loginOpen, setLoginOpen] = useState(false);
   523→  const activeQ = useQuery({
   524→    queryKey: ["active-user"],
   525→    queryFn: fetchActiveUser,
   526→  });
   527→  const isGuest = activeQ.data?.data?.id === "guest";
   528→
   529→  const usersQ = useQuery({
   530→    queryKey: ["users"],
   531→    queryFn: fetchUsers,
   532→    enabled: !isGuest && activeQ.data?.data?.permissions.includes("user:view") === true,
   533→  });
   534→
   535→  // Guest pakai endpoint publik (id, name, roleName only — no sensitive data)
   536→  const publicUsersQ = useQuery({
   537→    queryKey: ["users-public"],
   538→    queryFn: fetchPublicUsers,
   539→    enabled: isGuest,
   540→  });
   541→
   542→  const active = activeQ.data?.data ?? null;
   543→  const users = usersQ.data ?? [];
   544→  const publicUsers = publicUsersQ.data ?? [];
   545→  const hasUsers = activeQ.data?.hasUsers ?? false;
   546→  const canManageRoles = active?.permissions.includes("role:manage") ?? false;
   547→  const canViewUsers = active?.permissions.includes("user:view") ?? false;
   548→
   549→  // Buka login dialog dengan user terpilih
   550→  const openLogin = (user: PublicUser) => {
   551→    setLoginTarget(user);
   552→    setLoginOpen(true);
   553→  };
   554→
   555→  // Switch user (admin/editor yang sudah login) — juga perlu password
   556→  const switchMut = useMutation({
   557→    mutationFn: async (target: { id: string; name: string }) => {
   558→      // Bila target sama dengan aktif → skip
   559→      if (active?.id === target.id) return;
   560→      // Buka login dialog untuk user target (perlu password)
   561→      const pubUser = [...users, ...publicUsers].find((u) => u.id === target.id);
   562→      if (pubUser) {
   563→        setLoginTarget(pubUser);
   564→        setLoginOpen(true);
   565→      }
   566→    },
   567→  });
   568→
   569→  const logoutMut = useMutation({
   570→    mutationFn: async () => {
   571→      const res = await fetch("/api/users/active", { method: "DELETE" });
   572→      if (!res.ok) throw new Error("Gagal logout");
   573→    },
   574→    onSuccess: async () => {
   575→      await qc.invalidateQueries({ queryKey: ["active-user"] });
   576→      toast.success("Anda keluar — kembali sebagai Tamu (Viewer).");
   577→    },
   578→  });
   579→
   580→  const onLoggedIn = async () => {
   581→    await qc.invalidateQueries({ queryKey: ["active-user"] });
   582→    await qc.invalidateQueries({ queryKey: ["users"] });
   583→    await qc.invalidateQueries({ queryKey: ["users-public"] });
   584→  };
   585→
   586→  if (!active) return null;
   587→
   588→  return (
   589→    <>
   590→      <GuestOrUserMenu
   591→        active={active}
   592→        isGuest={isGuest}
   593→        hasUsers={hasUsers}
   594→        users={users}
